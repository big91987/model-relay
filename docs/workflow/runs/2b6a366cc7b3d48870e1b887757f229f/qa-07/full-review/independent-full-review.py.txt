from pathlib import Path
import collections, hashlib, json, os, re, subprocess, sys
root = Path.cwd()

run = Path('docs/workflow/runs') / run_id
archive = run / 'qa-07/full-review'
def sha(raw):
    return hashlib.sha256(raw).hexdigest()
def record(path):
    raw = path.read_bytes()
    return {'path': str(path), 'bytes': len(raw), 'sha256': sha(raw)}
def save(name, value):
    (archive / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')
head = subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip()

assert subprocess.check_output(['git', 'diff', 'HEAD', '--name-only']) == b''
subprocess.run(['git', 'diff', '--check'], check=True)
manifest = json.loads((run / 'first-round/source-manifest.json').read_text())
combined = hashlib.sha256()

    assert len(raw) == item['bytes'] and sha(raw) == item['sha256'], item['path']
    combined.update(f"{item['path']}\0{len(raw)}\0".encode())
    combined.update(raw)

receipts = [
    {'seq': 3, 'path': f'workflow-output/{run_id}/command-3.log', 'bytes': 1174629, 'sha256': '754f5242f9a567dd52b2f40b10667b34c18819099b6094d28940e48f661d310e', 'truncated': False},

reviews = []
for receipt in receipts:
    seq = receipt['seq']
    attempt = '20261014T192852Z-1089328441' if seq == 3 else '20261014T195523Z-3543396171'
    stage = 'qa-04/tests03' if seq == 3 else 'qa-07/tests06'
    outer_path = Path(receipt['path'])
    outer_raw = outer_path.read_bytes()
    assert len(outer_raw) == receipt['bytes'] and sha(outer_raw) == receipt['sha256']
    outer = outer_raw.decode('utf-8', errors='strict')
    inner_raw = (run / (stage + '.verify.log')).read_bytes()
    state_raw = (run / (stage + '.state.json')).read_bytes()
    state = json.loads(state_raw)
    assert inner_raw == (Path('test-results/verification') / attempt / 'verify.log').read_bytes()
    assert state_raw == (Path('test-results/verification') / attempt / 'state.json').read_bytes()
    assert len(inner_raw) == state['log_bytes'] and sha(inner_raw) == state['log_sha256']
    inner = inner_raw.decode('utf-8', errors='strict')
    lines = inner.splitlines()

    assert state['gate_configured_budget_ms'] == 1440000
    assert state['gate']['execute_ms'] < state['gate_configured_budget_ms']
    assert not state['gate']['cancelled'] and state['gate']['owned_group_absent']
    for key in ('clean_start_measured', 'source_unchanged', 'candidate_unchanged', 'owned_group_absent'):
        assert state[key]
    assert not state['node_modules_before'] and not state['npm_cache_before']
    assert state['manual_preinstall'] == 'none' and state['NODE_PATH'] == 'unset'
    assert 'candidate_removed' in state['cleanup']
    assert state['source_before'] == state['source_after'] and state['source_before']['head'] == head
    assert state['source_before']['working_tree_sha256'] == combined.hexdigest()
    assert all(item['exit'] == 0 and not item['cancelled'] and item['owned_group_absent'] for item in state['auxiliary_commands'])

    assert state['resources_before']['runtime_gomaxprocs'] == 2
    assert state['resource_delta']['counters']['memory.events.oom'] == 0
    assert state['resource_delta']['counters']['memory.events.oom_kill'] == 0
    events = re.findall(r'\{"Event":"(?:init|ping|execution_complete)","Data":"(?:\\.|[^"\\])*","ID":""\}\n?', outer)
    parsed_events = [json.loads(item) for item in events]
    event_counts = collections.Counter(item['Event'] for item in parsed_events)
    assert event_counts['init'] == event_counts['execution_complete'] == 1
    init = json.loads(next(item['Data'] for item in parsed_events if item['Event'] == 'init'))
    complete = json.loads(next(item['Data'] for item in parsed_events if item['Event'] == 'execution_complete'))
    assert init['text'] == ('48b9533c70bf4d6285ac76aca19dd530' if seq == 3 else '354f6de569b44a4fa596e8576fc55edb')

    assert 0 <= complete['execution_time'] - elapsed_wall_ms <= 1, 'millisecond quantization differs beyond 1ms'
    assert complete['execution_time'] < 1800000

        body = body.replace(event, '', 1)

    prefix = 'go run ./tests/verifyverification_receipt=test-results/verification/' + attempt
    assert normalize(body) == prefix + normalize(inner), 'full outer/inner transport comparison failed'
    assert '[... output omitted;' not in outer and 'chars truncated' not in outer
    assert '# tests 283' in inner and '# pass 283' in inner and '# fail 0' in inner
    assert 'all modules verified' in inner
    assert not any(line.startswith(('--- FAIL:', 'FAIL\t', 'FAIL ', 'not ok')) for line in lines)
    assert not re.search(r'WARNING: DATA RACE|^panic:', inner, re.M)

    for line in lines:
        if line.startswith('{"Time":'):

            test = preceding.removeprefix('=== RUN   ')

    nonzero_gates = []

        if line.startswith('GATE_END') and 'exit=0 ' not in line:

        assert any('model-relay/' + package + '\t' in line for line in packages)
