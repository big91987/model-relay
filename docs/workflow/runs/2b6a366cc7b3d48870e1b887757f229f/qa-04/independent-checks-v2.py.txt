from pathlib import Path
import datetime, hashlib, json, os, subprocess, sys
root = Path.cwd()
run = root / 'docs/workflow/runs/2b6a366cc7b3d48870e1b887757f229f'
archive = run / 'qa-04'
def sha(raw):
    return hashlib.sha256(raw).hexdigest()
manifest = json.loads((run / 'first-round/source-manifest.json').read_text())
combined = hashlib.sha256()
for item in sorted(manifest['files'], key=lambda entry: entry['path']):
    raw = (root / item['path']).read_bytes()
    assert len(raw) == item['bytes'] and sha(raw) == item['sha256'], item['path']
    combined.update(f"{item['path']}\0{len(raw)}\0".encode())
    combined.update(raw)
assert combined.hexdigest() == manifest['working_tree_sha256']
source = json.loads((root / '.harness/agent-platform/SOURCE.json').read_text())
for name, item in source['files'].items():
    assert sha((root / '.harness/agent-platform' / name).read_bytes()) == item['sha256'], name
subprocess.run(['sha256sum', '-c', str(run / 'SHA256SUMS')], check=True, stdout=subprocess.DEVNULL)
subprocess.run(['git', 'diff', '--check'], check=True)
assert subprocess.check_output(['git', 'diff', 'HEAD', '--name-only']) == b''
subprocess.run(['git', 'merge-base', '--is-ancestor', '6d2552d331fd7a0ec8d205f5d18ba3bcd34d6995', 'HEAD'], check=True)
for name in ('Makefile', 'docs/harness/opensandbox-run-sharing.md'):
    assert (root / name).read_bytes() == subprocess.check_output(['git', 'show', 'HEAD:' + name])
document = (root / 'docs/harness/opensandbox-publish-recovery.md').read_bytes()
assert document == (run / 'first-round/opensandbox-publish-recovery.md').read_bytes()
headings = [line for line in document.decode().splitlines() if line.startswith('#')]
missing = not any('停止' in line or '恢复与环境释放' in line for line in headings)
assert missing
state = json.loads((archive / 'tests03.state.json').read_text())
log = (archive / 'tests03.verify.log').read_bytes()
assert sha(log) == state['log_sha256'] and len(log) == state['log_bytes']
assert state['exit_code'] == state['gate_exit'] == 0
assert state['source_unchanged'] and state['candidate_unchanged'] and state['owned_group_absent']
assert state['clean_start_measured'] and not state['node_modules_before'] and not state['npm_cache_before']
assert state['gate_configured_budget_ms'] == 1440000
assert state['source_before'] == state['source_after']
assert state['source_before']['working_tree_sha256'] == combined.hexdigest()
assert 'candidate_removed' in state['cleanup']
decoded = log.decode()
assert not any(line.startswith('--- FAIL:') for line in decoded.splitlines())
assert '# fail 0' in decoded
assert 'PASS expected historical HTTP red: 21 method failures with exact status/Allow diagnostics; 12 query/recovery passes' in decoded
assert '--- PASS: TestFixedRunnerReceipts' in decoded
assert '--- PASS: TestInventoryCancellationReceipt' in decoded
for package in ('internal/relay', 'tests/verify', 'tests/health-history'):
    assert any(line.startswith('ok') and 'model-relay/' + package + '\t' in line for line in decoded.splitlines())
sys.path.insert(0, str(root / '.harness/agent-platform/examples/platform-workflows'))
from repository import check_publish_files
check_publish_files(root)
